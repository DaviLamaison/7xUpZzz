class carro:
    def __init__(self,modelo,velMax):
        self.modelo = modelo
        self.velAtual = 0
        self.velMax = velMax

    def acelerar(self, valor):
        velNova = self.velAtual + valor

        if velNova <= self.velMax:
            self.velAtual = velNova
        else:
            self.velAtual = self.velMax
            print("Velocidade maxima atingida")
    def frear(self,valor):
        velNova = self.velAtual - valor

        if velNova >= 0:
            self.velAtual = velNova
        else:
            self.velAtual = 0
            print("O carro parou compoletamente")

Carro = carro("Nissan GT-R", 300)

print("Modelo" ,Carro.modelo)
print("Velocidade inicial" ,Carro.velAtual ,"Km/h")

Carro.acelerar(100)
print("Após acelerar 100: " ,Carro.velAtual ,"Km/h")

Carro.acelerar(250)
print("Após tentar acelerar muito: " ,Carro.velAtual ,"Km/h")

Carro.frear(80)
print("Após  frear 80: " ,Carro.velAtual, "Km/h")
